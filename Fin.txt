import React, { useState, useEffect } from "react";
import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import * as z from "zod";
import "./App.css";

const schema = z.object({
  firstName: z
    .string()
    .min(1, { message: "Enter the first name" })
    .max(15, { message: "Name is too long" }),
  lastName: z
    .string()
    .min(1, { message: "Enter the last name" })
    .max(15, { message: "Name is too long" }),
  age: z.coerce
    .number()
    .min(18, { message: "Age must be above 18" })
    .max(60, { message: "60 is the maximum age limit" }),
  email: z.string().email("Email is invalid"),
  departmentId: z.coerce.number().min(1, { message: "Select a department" }),
  designationId: z.coerce.number().min(1, { message: "Select a designation" }),
  joiningDate: z.coerce.date(),
});

function App() {
  const [editingId, setEditingId] = useState(null);
  const [employees, setEmployees] = useState([]);
  const [departments, setDepartments] = useState([]);
  const [designations, setDesignations] = useState([]);
  const [selectedDepartmentId, setSelectedDepartmentId] = useState("");

  const {
    register,
    handleSubmit,
    formState: { errors },
    reset,
    setValue,
  } = useForm({ resolver: zodResolver(schema) });

  // Designations filtered by selected department
  const filteredDesignations = selectedDepartmentId
    ? designations.filter((d) => d.departmentId === Number(selectedDepartmentId))
    : [];

  // Fetch employees
  const fetchEmployees = async () => {
    try {
      const response = await fetch("http://localhost:3001/employees");
      if (!response.ok) throw new Error("Failed to load employees");
      const data = await response.json();
      setEmployees(data);
    } catch (error) {
      console.error("Error loading employees:", error);
    }
  };

  // Fetch departments
  const fetchDepartments = async () => {
    try {
      const response = await fetch("http://localhost:3001/departments");
      if (!response.ok) throw new Error("Failed to load departments");
      const data = await response.json();
      setDepartments(data);
    } catch (error) {
      console.error("Error loading departments:", error);
    }
  };

  // Fetch designations
  const fetchDesignations = async () => {
    try {
      const response = await fetch("http://localhost:3001/designations");
      if (!response.ok) throw new Error("Failed to load designations");
      const data = await response.json();
      setDesignations(data);
    } catch (error) {
      console.error("Error loading designations:", error);
    }
  };

  useEffect(() => {
    fetchEmployees();
    fetchDepartments();
    fetchDesignations();
  }, []);

  // Submit: Add or Update employee
  const onSubmit = async (data) => {
    try {
      const url = editingId
        ? `http://localhost:3001/employees/${editingId}`
        : "http://localhost:3001/employees";

      const method = editingId ? "PUT" : "POST";

      const response = await fetch(url, {
        method,
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(data),
      });

      if (!response.ok)
        throw new Error(`Failed to ${editingId ? "update" : "save"} employee`);

      alert(`Employee ${editingId ? "updated" : "added"} successfully!`);
      setEditingId(null);
      setSelectedDepartmentId("");
      reset();
      fetchEmployees();
    } catch (error) {
      console.error("Error:", error);
      alert(`Error ${editingId ? "updating" : "saving"} employee`);
    }
  };

  // Populate form for editing
  const handleEdit = (employee) => {
    setEditingId(employee.id);
    setSelectedDepartmentId(String(employee.departmentId)); // enable filtered designations
    setValue("firstName", employee.firstName);
    setValue("lastName", employee.lastName);
    setValue("age", employee.age);
    setValue("email", employee.email);
    setValue("departmentId", employee.departmentId);
    setValue("designationId", employee.designationId);
    setValue("joiningDate", employee.joiningDate);
  };

  // Delete employee
  const deleteEmployee = async (id) => {
    if (!window.confirm("Are you sure you want to delete this employee?"))
      return;
    try {
      const response = await fetch(`http://localhost:3001/employees/${id}`, {
        method: "DELETE",
      });
      if (!response.ok) throw new Error("Failed to delete employee");
      alert("Employee deleted successfully!");
      fetchEmployees();
    } catch (error) {
      console.error("Error deleting employee:", error);
      alert("Error deleting employee");
    }
  };

  // Helper: get department name by id
  const getDepartmentName = (departmentId) => {
    const dept = departments.find((d) => d.id === departmentId);
    return dept ? dept.name : "Unknown";
  };

  // Helper: get designation name by id
  const getDesignationName = (designationId) => {
    const desig = designations.find((d) => d.id === designationId);
    return desig ? desig.title : "Unknown";
  };

  return (
    <div>
      <div className="bg-gradient-to-r from-cyan-50 via-cyan-100 to-cyan-200 p-8 md:p-12 rounded-3xl shadow-xl text-center">
        <form
          className="max-w-md mx-auto bg-cyan-50 p-6 rounded shadow space-y-4"
          onSubmit={handleSubmit(onSubmit)}
        >
          <h2 className="text-xl font-mono font-bold text-cyan-800">
            {editingId ? "Edit Employee" : "Add Employee"}
          </h2>

          {/* FIRST NAME */}
          <label className="text-base font-mono">FIRST NAME</label>
          <input
            className="w-full bg-transparent placeholder:text-neutral-400 text-cyan-800 text-sm border border-slate-400 rounded-md px-3 py-2 transition duration-300 ease focus:outline-none focus:border-slate-50 hover:border-slate-300 shadow-sm"
            placeholder="Enter first name"
            {...register("firstName")}
          />
          {errors.firstName && <span className="text-red-500 text-sm">{errors.firstName.message}</span>}

          {/* LAST NAME */}
          <label className="text-base font-mono">LAST NAME</label>
          <input
            className="w-full bg-transparent placeholder:text-neutral-400 text-cyan-800 border border-slate-400 rounded-md px-3 py-2 transition duration-300 ease focus:outline-none focus:border-slate-50 hover:border-slate-300 shadow-sm"
            placeholder="Enter last name"
            {...register("lastName")}
          />
          {errors.lastName && <span className="text-red-500 text-sm">{errors.lastName.message}</span>}

          {/* AGE */}
          <label className="text-base font-mono">AGE</label>
          <input
            className="w-full bg-transparent placeholder:text-neutral-400 text-cyan-800 border border-slate-400 rounded-md px-3 py-2 transition duration-300 ease focus:outline-none focus:border-slate-50 hover:border-slate-300 shadow-sm"
            placeholder="Enter your age"
            type="number"
            {...register("age")}
          />
          {errors.age && <span className="text-red-500 text-sm">{errors.age.message}</span>}

          {/* EMAIL */}
          <label className="text-base font-mono">EMAIL</label>
          <input
            className="w-full bg-transparent placeholder:text-neutral-400 text-cyan-800 border border-slate-400 rounded-md px-3 py-2 transition duration-300 ease focus:outline-none focus:border-slate-50 hover:border-slate-300 shadow-sm"
            placeholder="Enter your email"
            {...register("email")}
          />
          {errors.email && <span className="text-red-500 text-sm">{errors.email.message}</span>}

          {/* DEPARTMENT — dropdown from db.json */}
          <label className="text-base font-mono">DEPARTMENT</label>
          <select
            className="w-full bg-white text-cyan-800 border border-slate-400 rounded-md px-3 py-2 transition duration-300 ease focus:outline-none hover:border-slate-300 shadow-sm"
            {...register("departmentId")}
            onChange={(e) => {
              setSelectedDepartmentId(e.target.value);
              setValue("designationId", ""); // reset designation when dept changes
            }}
          >
            <option value="">-- Select Department --</option>
            {departments.map((dept) => (
              <option key={dept.id} value={dept.id}>
                {dept.name}
              </option>
            ))}
          </select>
          {errors.departmentId && (
            <span className="text-red-500 text-sm">{errors.departmentId.message}</span>
          )}

          {/* DESIGNATION — filtered by selected department */}
          <label className="text-base font-mono">DESIGNATION</label>
          <select
            className="w-full bg-white text-cyan-800 border border-slate-400 rounded-md px-3 py-2 transition duration-300 ease focus:outline-none hover:border-slate-300 shadow-sm disabled:opacity-50 disabled:cursor-not-allowed"
            {...register("designationId")}
            disabled={!selectedDepartmentId}
          >
            <option value="">
              {selectedDepartmentId ? "-- Select Designation --" : "-- Select a Department first --"}
            </option>
            {filteredDesignations.map((desig) => (
              <option key={desig.id} value={desig.id}>
                {desig.title}
              </option>
            ))}
          </select>
          {errors.designationId && (
            <span className="text-red-500 text-sm">{errors.designationId.message}</span>
          )}

          {/* JOINING DATE */}
          <label className="text-base font-mono">JOINING DATE</label>
          <input
            className="w-full bg-transparent text-cyan-800 border border-slate-400 rounded-md px-3 py-2 transition duration-300 ease focus:outline-none focus:border-slate-50 hover:border-slate-300 shadow-sm"
            type="datetime-local"
            {...register("joiningDate")}
          />
          {errors.joiningDate && (
            <span className="text-red-500 text-sm">{errors.joiningDate.message}</span>
          )}

          {/* SUBMIT BUTTON */}
          <input
            className="w-full bg-cyan-600 text-white py-2 px-4 rounded hover:bg-cyan-900 transition cursor-pointer"
            type="submit"
            value={editingId ? "Update Employee" : "Add Employee"}
          />

          {/* CANCEL EDIT */}
          {editingId && (
            <button
              type="button"
              onClick={() => { setEditingId(null); setSelectedDepartmentId(""); reset(); }}
              className="w-full bg-gray-400 text-white py-2 px-4 rounded hover:bg-gray-600 transition"
            >
              Cancel
            </button>
          )}
        </form>

        {/* TABLE */}
        <div className="overflow-x-auto relative p-10">
          <table className="min-w-full text-sm text-left text-gray-700 font-mono">
            <thead className="text-xs text-white uppercase bg-cyan-600">
              <tr>
                <th className="px-6 py-3">FIRST NAME</th>
                <th className="px-6 py-3">LAST NAME</th>
                <th className="px-6 py-3">AGE</th>
                <th className="px-6 py-3">EMAIL</th>
                <th className="px-6 py-3">DEPARTMENT</th>
                <th className="px-6 py-3">DESIGNATION</th>
                <th className="px-6 py-3">JOINING DATE</th>
                <th className="px-6 py-3">ACTIONS</th>
              </tr>
            </thead>
            <tbody>
              {employees.map((emp) => (
                <tr className="bg-cyan-50 border-b border-gray-100" key={emp.id}>
                  <td className="px-6 py-4">{emp.firstName}</td>
                  <td className="px-6 py-4">{emp.lastName}</td>
                  <td className="px-6 py-4">{emp.age}</td>
                  <td className="px-6 py-4">{emp.email}</td>
                  <td className="px-6 py-4">{getDepartmentName(Number(emp.departmentId))}</td>
                  <td className="px-6 py-4">{getDesignationName(Number(emp.designationId))}</td>
                  <td className="px-6 py-4">{emp.joiningDate}</td>
                  <td className="px-6 py-4 space-y-1">
                    <button
                      onClick={() => handleEdit(emp)}
                      type="button"
                      className="w-full bg-cyan-600 text-white py-1 px-3 rounded hover:bg-cyan-900 transition"
                    >
                      EDIT
                    </button>
                    <button
                      onClick={() => deleteEmployee(emp.id)}
                      type="button"
                      className="w-full bg-red-500 text-white py-1 px-3 rounded hover:bg-red-700 transition"
                    >
                      DELETE
                    </button>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
}

export default App;
